cmake_minimum_required(VERSION 3.20)
project(kana_tsf LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "TSF integration requires Windows")
endif()

add_library(kana-tsf-preview SHARED tsf_probe.cpp)
target_compile_definitions(kana-tsf-preview PRIVATE KANA_LIVE_PREVIEW)
add_library(kana-tsf-edit SHARED tsf_edit_probe.cpp)
add_executable(tsf-probe-fixture tsf_probe_fixture.cpp)
foreach(target kana-tsf-preview kana-tsf-edit tsf-probe-fixture)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_link_libraries(${target} PRIVATE ole32 oleaut32 uuid user32)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8 /EHsc)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
