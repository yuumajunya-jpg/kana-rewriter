cmake_minimum_required(VERSION 3.20)
project(kana_editor_worker LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "The editor worker requires Windows")
endif()
add_executable(kana-editor-worker worker.cpp)
target_compile_features(kana-editor-worker PRIVATE cxx_std_17)
target_compile_definitions(kana-editor-worker PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(kana-editor-worker PRIVATE ole32 oleaut32 uuid user32 imm32)
if(MSVC)
  target_compile_options(kana-editor-worker PRIVATE /W4 /utf-8 /EHsc)
else()
  target_compile_options(kana-editor-worker PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME native-self-test COMMAND kana-editor-worker --self-test)
